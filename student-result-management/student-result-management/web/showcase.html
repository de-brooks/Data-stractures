Where arrays are used in the real world

Any time a program holds many values of the same kind and must visit them in order, an array is the first tool to reach for.

  • Student records

    Marks, attendance and fees for a class, processed together for totals and rankings.

  • Weather and sensor logs

    Hourly temperature readings stored in sequence to find highs, lows and trends.

  • Images and video

    A picture is a grid of pixel values; filters loop over every element.

  • Audio

    Sound is thousands of samples per second, stored and played back in order.

  • Spreadsheets and tables

    Rows and columns map to a two-dimensional array.

  • Games and boards

    Chess, tic-tac-toe and inventories track the state of every square or slot.

Why arrays are necessary

One name, many values

Declaring marks[5] replaces five separate variables and scales to five hundred.

Fast access by index

Elements sit side by side in memory, so marks[3] is found directly, without searching.

Loops fit naturally

A single for loop can total, compare and print every element.

Predictable memory

The size is fixed up front, so storage is compact and easy to reason about.

Chosen application: Student Result Management System

The program stores the marks of 5 students in one integer array, validates each entry, then finds the total, average, highest and lowest marks and labels each student Pass (50 or above) or Fail. Without an array it would need a separate variable for every student and repeated code for each one.

#include <iostream>
#include <iomanip>
using namespace std;

int main() {
    const int SIZE = 5;
    int marks[SIZE];
    int total = 0;
    int highest, lowest;
    double average;

    cout << "STUDENT RESULT MANAGEMENT SYSTEM\n";
    cout << "Enter marks for " << SIZE << " students (0-100).\n\n";

    // Input and validate students' marks
    for (int i = 0; i < SIZE; i++) {
        cout << "Enter mark for student " << i + 1 << ": ";
        cin >> marks[i];

        while (cin.fail() || marks[i] < 0 || marks[i] > 100) {
            cin.clear();
            cin.ignore(10000, '\\n');
            cout << "Invalid input. Enter a whole number from 0 to 100: ";
            cin >> marks[i];
        }
    }

    // Initialize highest and lowest using the first array element
    highest = marks[0];
    lowest = marks[0];

    // Process the array
    for (int i = 0; i < SIZE; i++) {
        total += marks[i];

        if (marks[i] > highest) {
            highest = marks[i];
        }

        if (marks[i] < lowest) {
            lowest = marks[i];
        }
    }

    average = static_cast<double>(total) / SIZE;

    // Display individual results
    cout << "\n--- STUDENT RESULTS ---\n";
    for (int i = 0; i < SIZE; i++) {
        cout << "Student " << i + 1 << ": " << marks[i] << " - "
             << (marks[i] >= 50 ? "Pass" : "Fail") << '\n';
    }

    // Display summary
    cout << fixed << setprecision(2);
    cout << "\nTotal marks: " << total << '\n';
    cout << "Average mark: " << average << '\n';
    cout << "Highest mark: " << highest << '\n';
    cout << "Lowest mark: " << lowest << '\n';

    return 0;
}

// Run with Node.js: node results.js
const readline = require("readline/promises");
const SIZE = 5;

async function main() {
  const rl = readline.createInterface({ input: process.stdin, output: process.stdout });
  const marks = new Array(SIZE);

  console.log("STUDENT RESULT MANAGEMENT SYSTEM");
  console.log(`Enter marks for ${SIZE} students (0-100).\n`);

  for (let i = 0; i < SIZE; i++) {
    let text = await rl.question(`Enter mark for student ${i + 1}: `);
    while (!/^\d+$/.test(text.trim()) || +text > 100) {
      text = await rl.question("Invalid input. Enter a whole number from 0 to 100: ");
    }
    marks[i] = Number(text);
  }
  rl.close();

  let total = 0, highest = marks[0], lowest = marks[0];
  for (const m of marks) {
    total += m;
    if (m > highest) highest = m;
    if (m < lowest) lowest = m;
  }
  const average = total / SIZE;

  console.log("\n--- STUDENT RESULTS ---");
  marks.forEach((m, i) => console.log(`Student ${i + 1}: ${m} - ${m >= 50 ? "Pass" : "Fail"}`));

  console.log(`\nTotal marks: ${total}`);
  console.log(`Average mark: ${average.toFixed(2)}`);
  console.log(`Highest mark: ${highest}`);
  console.log(`Lowest mark: ${lowest}`);
}
main();

import java.util.Scanner;

public class StudentResults {
    public static void main(String[] args) {
        final int SIZE = 5;
        int[] marks = new int[SIZE];
        int total = 0;
        Scanner in = new Scanner(System.in);

        System.out.println("STUDENT RESULT MANAGEMENT SYSTEM");
        System.out.println("Enter marks for " + SIZE + " students (0-100).\n");

        for (int i = 0; i < SIZE; i++) {
            System.out.print("Enter mark for student " + (i + 1) + ": ");
            while (true) {
                if (in.hasNextInt()) {
                    marks[i] = in.nextInt();
                    if (marks[i] >= 0 && marks[i] <= 100) break;
                } else {
                    in.next();
                }
                System.out.print("Invalid input. Enter a whole number from 0 to 100: ");
            }
        }

        int highest = marks[0], lowest = marks[0];
        for (int m : marks) {
            total += m;
            if (m > highest) highest = m;
            if (m < lowest) lowest = m;
        }
        double average = (double) total / SIZE;

        System.out.println("\n--- STUDENT RESULTS ---");
        for (int i = 0; i < SIZE; i++) {
            System.out.println("Student " + (i + 1) + ": " + marks[i] + " - "
                + (marks[i] >= 50 ? "Pass" : "Fail"));
        }

        System.out.println("\nTotal marks: " + total);
        System.out.printf("Average mark: %.2f%n", average);
        System.out.println("Highest mark: " + highest);
        System.out.println("Lowest mark: " + lowest);
    }
}
